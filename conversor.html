<script>

(() => {

  const nombres = {2:'binario',8:'octal',10:'decimal',16:'hexadecimal'};

  const permitidas = {2:/^+$/i, 8:/^[0-7]+$/i, 10:/^[0-9]+$/i, 16:/^[0-9a-f]+$/i};

  const entrada = document.getElementById('entrada');

  const hint = document.getElementById('hint');

  const pesosDiv = document.getElementById('pesos');

  const tarjetas = [...document.querySelectorAll('.out')];

  let base = 10, agrupa = false;


  // ---------- Utilidades con BigInt (sin límite práctico de tamaño) ----------

  function limpia(txt){

    return txt.trim().replace(/^0[xob]/i,'').replace(/\s|_/g,'');

  }

  function aBigInt(txt, b){

    if(!txt) return null;

    if(!permitidas[b].test(txt.toLowerCase())) return null;

    try { return BigInt(parseInt(b) + 'n' === '' ? 0 : ('0x'+'')); } catch(e){}

    // Conversión manual segura posición a posición

    const dig = txt.toUpperCase();

    let n = 0n, B = BigInt(b);

    for(const c of dig){

      const v = parseInt(c, 36);

      if(Number.isNaN(v) || v >= b) return null;

      n = n * B + BigInt(v);

    }

    return n;

  }

  function desdeBase(txt, b){ return aBigInt(txt, b); }

  function formato(n, b){

    switch(b){

      case 2:  { let s = n.toString(2); return agrupa ? s.replace(/(\d{1,4})(?=\d)/g,'$1 ').trim() : s; }

      case 8:  return n.toString(8);

      case 10: return n.toString(10).replace(/\B(?=(\d{3})+(?!\d))/g,' ');

      case 16: return n.toString(16).toUpperCase();

    }

  }

  function bruto(n,b){ return b===2 ? n.toString(2) : b===8 ? n.toString(8) : b===10 ? n.toString(10) : n.toString(16).toUpperCase(); }


  // ---------- Descomposición por pesos ----------

  function descomp(txt, b){

    if(!txt) return '';

    const digs = txt.toUpperCase().split('');

    const L = digs.length;

    const partes = digs.map((c,i) => {

      const v = parseInt(c,36), p = L-1-i;

      return `${c}·${b}^${p}`;

    });

    const total = partes.join('  +  ');

    const valor = desdeBase(txt,b);

    return `Entrada ${nombres[b]}:  ${total}\n= ${valor === null ? '?' : valor.toString(10)} (decimal)`;

  }


  // ---------- Render ----------

  function render(){

    const raw = limpia(entrada.value);

    entrada.classList.remove('invalid');

    hint.classList.remove('err');


    if(raw === ''){

      tarjetas.forEach(t => { t.querySelector('.val').textContent='—'; t.classList.remove('is-src'); });

      hint.textContent = 'Escribe un número en la base seleccionada arriba.';

      pesosDiv.textContent = '';

      return;

    }

    const n = desdeBase(raw, base);

    if(n === null){

      entrada.classList.add('invalid');

      hint.classList.add('err');

      hint.textContent = `Carácter no válido para base ${base} (${nombres[base]}). Cifras permitidas: ${cifras(base)}.`;

      tarjetas.forEach(t => t.classList.remove('is-src'));

      pesosDiv.textContent = '';

      return;

    }

    tarjetas.forEach(t => {

      const b = Number(t.dataset.base);

      t.querySelector('.val').textContent = formato(n, b);

      t.classList.toggle('is-src', b === base);

    });

    hint.textContent = `${formato(n,10)} en decimal · ${raw.length} cifra(s) en ${nombres[base]}`;

    pesosDiv.textContent = descomp(raw, base);

  }

  function cifras(b){ return b===2?'0–1':b===8?'0–7':b===10?'0–9':'0–9, A–F'; }


  // ---------- Eventos ----------

  document.getElementById('bases').addEventListener('click', e => {

    const btn = e.target.closest('button'); if(!btn) return;

    base = Number(btn.dataset.base);

    [...e.currentTarget.children].forEach(b => b.classList.toggle('active', b === btn));

    entrada.placeholder = base===16 ? 'Ej.: 1F4' : base===2 ? 'Ej.: 111 1110 010' : base===8 ? 'Ej.: 3752' : 'Ej.: 2026';

    render();

  });

  entrada.addEventListener('input', render);

  document.getElementById('clear').addEventListener('click', () => { entrada.value=''; entrada.focus(); render(); });

  document.getElementById('grpBin').addEventListener('click', e => {

    agrupa = !agrupa;

    e.target.textContent = agrupa ? 'Quitar separación en nibbles' : 'Agrupar binario en nibbles';

    render();

  });

  document.getElementById('swap').addEventListener('click', () => {

    const origen = tarjetas.find(t => Number(t.dataset.base) !== base);

    if(!origen) return;

    const b = Number(origen.dataset.base);

    const val = origen.querySelector('.val').textContent.replace(/\s/g,'');

    if(val === '—') return;

    entrada.value = val; base = b;

    [...document.getElementById('bases').children].forEach(x => x.classList.toggle('active', Number(x.dataset.base)===b));

    render();

  });

  document.querySelectorAll('.copy').forEach(btn => {

    btn.addEventListener('click', async () => {

      const txt = btn.parentElement.querySelector('.val').textContent.replace(/\s/g,'');

      if(txt === '—') return;

      try { await navigator.clipboard.writeText(txt); }

      catch(e){ const ta=document.createElement('textarea'); ta.value=txt; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); ta.remove(); }

      btn.textContent = '✓ copiado'; btn.classList.add('done');

      setTimeout(()=>{ btn.textContent='copiar'; btn.classList.remove('done'); }, 1200);

    });

  });

  document.addEventListener('keydown', e => { if(e.key === 'Escape'){ entrada.value=''; render(); } });


  render();

})();

</script>
